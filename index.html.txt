<script>
    let timer;
    let timeLeft = 25 * 60;
    let isRunning = false;
    let currentMode = 'work';
    let completedPomodoros = 0;

    const timerDisplay = document.getElementById('timer');
    const startBtn = document.getElementById('startBtn');
    const completedCountDisplay = document.getElementById('completedCount');

    function updateDisplay() {
        const minutes = Math.floor(timeLeft / 60);
        const seconds = timeLeft % 60;
        timerDisplay.textContent = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
    }

    function toggleTimer() {
        if (isRunning) {
            clearInterval(timer);
            startBtn.textContent = '繼續';
            isRunning = false;
        } else {
            isRunning = true;
            startBtn.textContent = '暫停';
            timer = setInterval(() => {
                if (timeLeft > 0) {
                    timeLeft--;
                    updateDisplay();
                } else {
                    clearInterval(timer);
                    isRunning = false;
                    alert(currentMode === 'work' ? '🎉 專注時間結束！休息一下吧！' : '💪 休息結束，準備開始專注！');
                    if (currentMode === 'work') {
                        completedPomodoros++;
                        completedCountDisplay.textContent = completedPomodoros;
                    }
                    resetTimer();
                }
            }, 1000);
        }
    }

    function resetTimer() {
        clearInterval(timer);
        isRunning = false;
        startBtn.textContent = '開始';
        timeLeft = currentMode === 'work' ? 25 * 60 : 5 * 60;
        updateDisplay();
    }

    function setMode(mode) {
        currentMode = mode;
        document.getElementById('workMode').classList.toggle('active', mode === 'work');
        document.getElementById('restMode').classList.toggle('active', mode === 'rest');
        document.documentElement.style.setProperty('--primary', mode === 'work' ? '#ff6b6b' : '#4ecdc4');
        resetTimer();
    }

    function addTask() {
        const input = document.getElementById('taskInput');
        const text = input.value.trim();
        if (!text) return;

        const li = document.createElement('li');
        li.innerHTML = `
            <span onclick="this.parentElement.classList.toggle('completed')">${text}</span>
            <button class="delete-btn" onclick="this.parentElement.remove()">✕</button>
        `;
        document.getElementById('taskList').appendChild(li);
        input.value = '';
    }
</script>
